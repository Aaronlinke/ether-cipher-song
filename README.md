function b58encode(bytes) {
  let x = BigInt("0x" + [...bytes].map(b=>b.toString(16).padStart(2,"0")).join(""));
  let out = "";
  while (x > 0n) {
    const mod = x % 58n;
    out = ALPHABET[Number(mod)] + out;
    x = x / 58n;
  }
  // leading zeros
  let leading = 0;
  for (const b of bytes) { if (b === 0) leading++; else break; }
  return "1".repeat(leading) + out;
}
function b58decode(str) {
  let x = 0n;
  for (const ch of str) x = x * 58n + BigInt(ALPHABET.indexOf(ch));
  let hex = x.toString(16);
  if (hex.length % 2) hex = "0" + hex;

  let leading = 0;

    z.set(bytes, leading);
    bytes = z;
  }
  return bytes;
}

function hexToBytes(hex){
  hex = hex.trim().toLowerCase();
  if (hex.startsWith("0x")) hex = hex.slice(2);
  if (!/^[0-9a-f]{64}$/.test(hex)) throw new Error("HEX muss 64 Zeichen sein (32 Bytes).");
  const out = new Uint8Array(32);
  for (let i=0;i<32;i++) out[i] = parseInt(hex.slice(i*2,i*2+2),16);
  return out;
}

async function hash160(bytes){

async function base58check(versionByte, payload){
  const data = new Uint8Array(1 + payload.length);
  data[0] = versionByte;
  data.set(payload, 1);
  const c1 = await sha256(data);
  const c2 = await sha256(c1);
  const chk = c2.slice(0,4);
  const out = new Uint8Array(data.length + 4);
  out.set(data,0); out.set(chk,data.length);
  return b58encode(out);
}

function mod(a,m){ const r=a%m; return r>=0n?r:r+m; }
function inv(a){ // a^(p-2) mod p
  let e = P-2n, base = mod(a,P), res=1n;
  while(e>0n){ if(e&1n) res=mod(res*base,P); base=mod(base*base,P); e>>=1n; }
  return res;
}
function pointAdd(A,B){
  if(!A) return B;
  if(!B) return A;
  const [x1,y1]=A, [x2,y2]=B;
  if(x1===x2 && y1!==y2) return null;
  let m;
  if(x1===x2){
    m = mod(3n*x1*x1 * inv(2n*y1), P);
  } else {
    m = mod((y1 - y2) * inv(x1 - x2), P);
  }
  const x3 = mod(m*m - x1 - x2, P);
  const y3 = mod(m*(x1 - x3) - y1, P);
  return [x3,y3];
}
function pointMul(k){
  let n = k;
  let Q = null;
  let R = [Gx, Gy];
  while(n>0n){
    if(n&1n) Q = pointAdd(Q,R);
    R = pointAdd(R,R);
    n >>= 1n;
  }
  return Q;
}

function wifToPrivBytes(wif){
  const raw = b58decode(wif.trim());
  if (raw.length < 4) throw new Error("WIF zu kurz.");
  const data = raw.slice(0, raw.length-4);
  // checksum check
  // (quick verify)
  // We skip deep validation in UI; this is a proof tool.
  const version = data[0];
  if (version !== 0x80) throw new Error("Nicht Mainnet-WIF (Version != 0x80).");
  let key = data.slice(1);
  let compressed = false;
  if (key.length === 33 && key[key.length-1] === 0x01) {
    compressed = true;
    key = key.slice(0,32);
  } else if (key.length !== 32) {
    throw new Error("Unerwartete WIF-Key-Länge.");
  }
  return { key, compressed };
}

async function privToAddresses(privBytes){
  const k = BigInt("0x"+bytesToHex(privBytes));
  const pub = pointMul(k);
  if(!pub) throw new Error("Ungültiger Key (Punkt im Unendlichen).");

  const x = pub[0], y = pub[1];

  // uncompressed pubkey
  const pubU = new Uint8Array(65);
  pubU[0]=0x04;
  const xBytes = hexToBytes(x.toString(16).padStart(64,"0"));
  const yBytes = hexToBytes(y.toString(16).padStart(64,"0"));
  pubU.set(xBytes,1); pubU.set(yBytes,33);

  // compressed pubkey
  const pubC = new Uint8Array(33);
  pubC[0] = (y % 2n === 0n) ? 0x02 : 0x03;
  pubC.set(xBytes,1);

  const h160U = await hash160(pubU);
  const h160C = await hash160(pubC);

  const addrU = await base58check(0x00, h160U);
  const addrC = await base58check(0x00, h160C);

  return { addrU, addrC };
}

        class Neuron {
            constructor(id, layer, index) {
                this.id = id;
                this.layer = layer;
                this.index = index;
                this.potential = 0;
                this.threshold = 1.0;
                this.decay = 0.95; // Energy loss over time
                this.weights = {}; // Connection strengths to other neurons
                this.lastFired = 0;
                this.refractoryPeriod = 5; // Ticks until can fire again
                this.refractoryCount = 0;
            }

            connect(targetNeuronId, weight) {
                this.weights[targetNeuronId] = weight;
            }

            receive(signal) {
                this.potential += signal;
            }

            tick() {
                if (this.refractoryCount > 0) {
                    this.refractoryCount--;
                    this.potential = 0; // Reset during refractory
                    return { fired: false, output: 0, potential: this.potential };
                }

                this.potential *= this.decay; // Natural decay

                if (this.potential >= this.threshold) {
                    const output = this.potential; // Sigmoid or similar could be applied here
                    this.potential = 0;
                    this.refractoryCount = this.refractoryPeriod;
                    this.lastFired = Date.now();
                    return { fired: true, output: output, potential: 1.0 }; // Visuals use potential 1.0 for fire
                }
                return { fired: false, output: 0, potential: this.potential };
